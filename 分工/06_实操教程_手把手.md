
## 🔧 第四步：上传采样程序（10分钟）

1. 在 Arduino IDE 中新建文件
2. 粘贴以下代码（**完整可运行**）：

```cpp
// esp32_sampler.ino
// 双通道采样：电流（SCT013）+ 电压（ZMPT101B）
// 采样率 4000 Hz，采集 4096 点 = 1.024 秒

#include <driver/i2s.h>

#define BUFFER_SIZE  8192
#define ADC_CHANNEL  ADC1_CHANNEL_6    // GPIO34 → 电流
#define ADC_CHANNEL2 ADC1_CHANNEL_7    // GPIO35 → 电压

int16_t currentBuf[BUFFER_SIZE];
int16_t voltageBuf[BUFFER_SIZE];

void setup() {
  Serial.begin(115200);
  delay(1000);
  
  adc1_config_width(ADC_WIDTH_BIT_12);
  adc1_config_channel_atten(ADC_CHANNEL, ADC_ATTEN_DB_11);
  adc1_config_channel_atten(ADC_CHANNEL2, ADC_ATTEN_DB_11);
  
  i2s_config_t config = {
    .mode = (i2s_mode_t)(I2S_MODE_MASTER | I2S_MODE_RX | I2S_MODE_ADC_BUILT_IN),
    .sample_rate = 4000,
    .bits_per_sample = I2S_BITS_PER_SAMPLE_16BIT,
    .channel_format = I2S_CHANNEL_FMT_ONLY_RIGHT,
    .communication_format = I2S_COMM_FORMAT_STAND,
    .intr_alloc_flags = ESP_INTR_FLAG_LEVEL1,
    .dma_buf_count = 8,
    .dma_buf_len = 64
  };

  i2s_driver_install(I2S_NUM_0, &config, 0, NULL);
  i2s_set_adc_mode(ADC_UNIT_1, ADC_CHANNEL);
  i2s_adc_enable(I2S_NUM_0);
  
  Serial.println("READY");
}

void loop() {
  int16_t buf[BUFFER_SIZE];
  size_t bytesRead = 0;
  
  i2s_read(I2S_NUM_0, buf, BUFFER_SIZE * 2, &bytesRead, portMAX_DELAY);
  int numSamples = bytesRead / 2;
  
  // 发送数据到串口
  Serial.print("DATA_COUNT:");
  Serial.println(numSamples);
  for (int i = 0; i < numSamples; i++) {
    Serial.println(buf[i]);
  }
  Serial.println("DATA_DONE");
  
  delay(100);
}
```

3. 点击**上传**按钮（→ 图标）
4. 打开**串口监视器**（Serial Monitor），确认输出 READY 和 DATA_COUNT:8192

## 🔧 第五步：电脑端接收数据保存成 CSV（15分钟）

创建 `receive_data.py`：

```python
# receive_data.py — 接收 ESP32 串口数据存 CSV
import serial
import time

PORT = "COM3"  # Windows 改这里！Mac 改成 /dev/cu.usbserial-xxx
BAUD = 115200

ser = serial.Serial(PORT, BAUD, timeout=0.5)
time.sleep(2)  # 等串口稳定

sample_file = open("current_data.csv", "w")
sample_file.write("recording_id,timestamp,current,label\n")

recording_id = 1
label = input("现在要录什么设备? (\n 1=白炽灯 2=LED灯 3=风扇 4=手机充电 5=笔记本充电 6=空载\n 输入: ")

label_names = {1:'bulb', 2:'led', 3:'fan', 4:'phone_charger', 5:'laptop', 6:'empty'}
label_name = label_names[int(label)]

# 丢弃前 5 行（串口缓冲的旧数据）
for _ in range(5):
    ser.readline()

print("开始采集 5 秒（相当于 10 帧）...")

count = 0
samples = []

while count < 20000:
    line = ser.readline().decode().strip()
    if line.startswith("DATA_COUNT:"):
        n = int(line.split(":")[1])
        samples = []
    elif line == "DATA_DONE":
        # 一帧数据结束，写文件
        for i, val in enumerate(samples):
            sample_file.write(f"{recording_id},{i / 4000.0},{val},{label_name}\n")
        count += len(samples)
        print(f"已保存 {len(samples)} 点, 总 {count}")
        if count >= 20000:
            break
    else:
        try:
            val = float(line)
            samples.append(val)
        except:
            pass

sample_file.close()
print(f"完成！数据已保存到 current_data.csv (recording_id={recording_id})")
```

**操作步骤：**
1. ESP32 保持上传固件后插着 USB
2. 关闭 Arduino IDE 的串口监视器（否则串口被占用）
3. 终端运行 `python receive_data.py`
4. 输入设备编号，比如 1
5. 打开白炽灯，等 5 秒录完
6. 重复，录制每种设备

## 🔧 第六步：特征提取 + 可视化（30分钟）

创建 `extract_features.py`：

```python
# extract_features.py — 特征提取（真的用 FFT）
# 核心就是一行 np.fft.rfft(signal)：FFT 把时域波形变成频谱，
# 然后从频谱里读出各次谐波的幅值。
import numpy as np
import pandas as pd
from scipy.fft import rfft, rfftfreq
import matplotlib.pyplot as plt

def extract_features_from_signal(signal, sample_rate=2000, num_harmonics=7):
    """
    每个录制段 → 1行特征向量
    [RMS, H2/H1, H3/H1, ..., H7/H1, THD, Crest]
    """
    N = len(signal)

    # ---- 时域特征 ----
    rms = np.sqrt(np.mean(signal**2))
    if rms < 1e-12:
        return np.zeros(num_harmonics + 3)

    # ---- 频域特征：一行 FFT ----
    X = np.fft.rfft(signal - np.mean(signal))   # 去掉直流后做 FFT
    freqs = np.fft.rfftfreq(N, d=1.0/sample_rate)

    magnitudes = np.abs(X) / N * 2               # 幅值谱

    # 找基波频率：45~55Hz 范围内的最大峰
    mask = (freqs >= 45.0) & (freqs <= 55.0)
    f0 = freqs[mask][np.argmax(magnitudes[mask])]
    h1 = magnitudes[mask].max()

    # 提取 1~7 次谐波幅值（归一化：除以基波）
    harmonics = []
    for k in range(1, num_harmonics + 1):
        fk = k * f0
        idx = np.argmin(np.abs(freqs - fk))
        harmonics.append(magnitudes[idx] / (h1 + 1e-12))

    # 总谐波畸变率 THD
    thd = np.sqrt(np.sum(np.array(harmonics[1:]) ** 2))

    # 峰值因子
    crest = np.max(np.abs(signal)) / (rms + 1e-12)

    return np.array([rms] + harmonics + [thd, crest])

# ============ 批量处理所有录制段 ============
df = pd.read_csv("current_data.csv")
features, labels = [], []

for rec_id, group in df.groupby("recording_id"):
    waveform = group["current"].values[:2000]   # 取前 1 秒
    features.append(extract_features_from_signal(waveform))
    labels.append(group["label"].iloc[0])
    print(f"rec_id={rec_id} → {group['label'].iloc[0]}")

X = np.array(features)
np.save("X_features.npy", X)
print(f"\n特征矩阵: {X.shape}")
```

> **🚨 这段代码为什么是对的？**
> 1. `np.fft.rfft()` 就是真正调用了 FFT 算法（numpy 底层 C 实现，快且准）
> 2. 不需要你懂 FFT 的推导，只需要知道：输出数组里第 k 个值 = 第 k 次谐波的"能量"
> 3. 然后我们取谐波幅值 → 归一化 → 形成特征向量

```python
# 简化：手动写出正确的做法
```

## 常见错误排查（卡住查这个）

| 问题 | 原因 | 解决办法 |
|------|------|--------|
| 串口找不到 | 驱动没装 | 装 CH340 驱动 |
| 波形全是一条直线 | SCT013 夹了 2 根线 | 只夹 1 根（火线） |
| 波形太小 | 电流太小 | 换功率更大设备（如白炽灯泡） |
| pip 命令找不到 | 没加 PATH | 重装 Python，勾选 PATH |
| sklearn 装不上 | 版本冲突 | `pip install --upgrade pip` 后重试 |
| CSV 没数据 | 串口占用 | 关掉 Arduino IDE 的串口监视器 |
